import lombok.RequiredArgsConstructor;
import org.springframework.core.io.Resource;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.multipart.MultipartFile;
import ru.university.jewelry.dto.JewelryFilter;
import ru.university.jewelry.dto.JewelryRequest;
import ru.university.jewelry.dto.JewelryResponse;
import ru.university.jewelry.exception.NotFoundException;
import ru.university.jewelry.model.Category;
import ru.university.jewelry.model.Jewelry;
import ru.university.jewelry.repository.CategoryRepository;
import ru.university.jewelry.repository.JewelryRepository;
import ru.university.jewelry.repository.spec.JewelrySpecifications;
import ru.university.jewelry.storage.StorageService;

import java.util.List;

@Service
@RequiredArgsConstructor
public class JewelryService {

    private final JewelryRepository jewelryRepository;
    private final CategoryRepository categoryRepository;
    private final StorageService storageService;

    @Transactional(readOnly = true)
    public List<JewelryResponse> findAll(JewelryFilter filter) {
        return jewelryRepository.findAll(JewelrySpecifications.withFilter(filter))
                .stream()
                .map(this::mapToResponse)
                .toList();
    }

    @Transactional(readOnly = true)
    public JewelryResponse findById(Long id) {
        return mapToResponse(getEntityById(id));
    }

    @Transactional
    public JewelryResponse create(JewelryRequest request) {
        Category category = categoryRepository.findById(request.getCategoryId())
                .orElseThrow(() -> new NotFoundException("Категория не найдена: " + request.getCategoryId()));

        Jewelry jewelry = Jewelry.builder()
                .name(request.getName())
                .description(request.getDescription())
                .metal(request.getMetal())
                .weight(request.getWeight())
                .price(request.getPrice())
                .status(request.getStatus())
                .category(category)
                .build();

        return mapToResponse(jewelryRepository.save(jewelry));
    }

    @Transactional
    public JewelryResponse update(Long id, JewelryRequest request) {
        Jewelry jewelry = getEntityById(id);
        Category category = categoryRepository.findById(request.getCategoryId())
                .orElseThrow(() -> new NotFoundException("Категория не найдена: " + request.getCategoryId()));

        jewelry.setName(request.getName());
        jewelry.setDescription(request.getDescription());
        jewelry.setMetal(request.getMetal());
        jewelry.setWeight(request.getWeight());
        jewelry.setPrice(request.getPrice());
        jewelry.setStatus(request.getStatus());
        jewelry.setCategory(category);

        return mapToResponse(jewelryRepository.save(jewelry));
    }

    @Transactional
    public void delete(Long id) {
        Jewelry jewelry = getEntityById(id);
        if (jewelry.getImagePath() != null) {
            storageService.delete(jewelry.getImagePath());
        }
        jewelryRepository.delete(jewelry);
    }

    @Transactional
    public JewelryResponse uploadPhoto(Long id, MultipartFile file) {
        Jewelry jewelry = getEntityById(id);
        if (jewelry.getImagePath() != null) {
            storageService.delete(jewelry.getImagePath());
        }
        String filename = storageService.store(file);
        jewelry.setImagePath(filename);
        return mapToResponse(jewelryRepository.save(jewelry));
    }

    @Transactional(readOnly = true)
    public Resource getPhoto(Long id) {
        Jewelry jewelry = getEntityById(id);
        if (jewelry.getImagePath() == null) {
            throw new NotFoundException("У изделия нет загруженного фото");
        }
        return storageService.loadAsResource(jewelry.getImagePath());
    }
